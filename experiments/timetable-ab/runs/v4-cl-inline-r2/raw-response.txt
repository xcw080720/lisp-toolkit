(defstruct (course (:constructor make-course (name teacher loc day from to)))
  (name "" :type string)
  (teacher "" :type string)
  (loc "" :type string)
  (day 1 :type integer)
  (from 1 :type integer)
  (to 1 :type integer))

(defvar *courses* '())

(defun split-words (s)
  (let ((words '())
        (start 0)
        (len (length s)))
    (labels ((whitespace-p (c)
               (or (char= c #\Space)
                   (char= c #\Tab)
                   (char= c #\Newline)
                   (char= c #\Return))))
      (loop for i from 0 below len
            do (when (whitespace-p (char s i))
                 (when (> i start)
                   (push (subseq s start i) words))
                 (setf start (1+ i))))
      (when (> len start)
        (push (subseq s start len) words))
      (nreverse words))))

(defun valid-word-p (s)
  (and (stringp s) (>= (length s) 1) (<= (length s) 12)))

(defun parse-int (s)
  (ignore-errors (parse-integer s :junk-allowed nil)))

(defun find-course (name)
  (find name *courses* :key #'course-name :test #'string=))

(defun course-overlaps-p (day from to &optional ignore-name)
  (loop for c in *courses*
        when (and (or (null ignore-name) (not (string= (course-name c) ignore-name)))
                  (= (course-day c) day)
                  (not (or (< (course-to c) from)
                           (< to (course-from c)))))
          return t))

(defun course-name< (a b)
  (let ((name-a (course-name a))
        (name-b (course-name b)))
    (let ((len (min (length name-a) (length name-b))))
      (loop for i from 0 below len
            for ca = (char-code (char name-a i))
            for cb = (char-code (char name-b i))
            when (< ca cb) return t
            when (> ca cb) return nil
            finally (return (< (length name-a) (length name-b)))))))

(defun handle-add (args)
  (if (/= (length args) 6)
      (format t "err BADARG~%")
      (let ((name (nth 0 args))
            (teacher (nth 1 args))
            (loc (nth 2 args))
            (day-s (nth 3 args))
            (from-s (nth 4 args))
            (to-s (nth 5 args)))
        (let ((day (parse-int day-s))
              (from (parse-int from-s))
              (to (parse-int to-s)))
          (if (or (not (valid-word-p name))
                  (not (valid-word-p teacher))
                  (not (valid-word-p loc))
                  (not day) (< day 1) (> day 7)
                  (not from) (< from 1) (> from 12)
                  (not to) (< to 1) (> to 12)
                  (< to from))
              (format t "err BADARG~%")
              (cond ((find-course name) (format t "err DUP~%"))
                    ((>= (length *courses*) 20) (format t "err LIMIT~%"))
                    ((course-overlaps-p day from to) (format t "err CONFLICT~%"))
                    (t
                     (push (make-course name teacher loc day from to) *courses*)
                     (format t "ok add 1~%"))))))))

(defun handle-del (args)
  (if (/= (length args) 1)
      (format t "err BADARG~%")
      (let ((name (first args)))
        (if (not (valid-word-p name))
            (format t "err BADARG~%")
            (let ((c (find-course name)))
              (if c
                  (progn (setf *courses* (remove c *courses* :test #'eq))
                         (format t "ok del 1~%"))
                  (format t "err NOTFOUND~%")))))))

(defun handle-edit (args)
  (if (/= (length args) 3)
      (format t "err BADARG~%")
      (let ((name (nth 0 args))
            (teacher (nth 1 args))
            (loc (nth 2 args)))
        (if (or (not (valid-word-p name))
                (not (valid-word-p teacher))
                (not (valid-word-p loc)))
            (format t "err BADARG~%")
            (let ((c (find-course name)))
              (if c
                  (progn (setf (course-teacher c) teacher
                               (course-loc c) loc)
                         (format t "ok edit 1~%"))
                  (format t "err NOTFOUND~%")))))))

(defun handle-move (args)
  (if (/= (length args) 4)
      (format t "err BADARG~%")
      (let ((name (nth 0 args))
            (day-s (nth 1 args))
            (from-s (nth 2 args))
            (to-s (nth 3 args)))
        (let ((day (parse-int day-s))
              (from (parse-int from-s))
              (to (parse-int to-s)))
          (if (or (not (valid-word-p name))
                  (not day) (< day 1) (> day 7)
                  (not from) (< from 1) (> from 12)
                  (not to) (< to 1) (> to 12)
                  (< to from))
              (format t "err BADARG~%")
              (let ((c (find-course name)))
                (if c
                    (if (course-overlaps-p day from to name)
                        (format t "err CONFLICT~%")
                        (progn (setf (course-day c) day
                                     (course-from c) from
                                     (course-to c) to)
                               (format t "ok move 1~%")))
                    (format t "err NOTFOUND~%"))))))))

(defun handle-swap (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((a-name (nth 0 args))
            (b-name (nth 1 args)))
        (if (or (not (valid-word-p a-name))
                (not (valid-word-p b-name)))
            (format t "err BADARG~%")
            (let ((a (find-course a-name))
                  (b (find-course b-name)))
              (if (and a b)
                  (let ((a-day (course-day a))
                        (a-from (course-from a))
                        (a-to (course-to a)))
                    (setf (course-day a) (course-day b)
                          (course-from a) (course-from b)
                          (course-to a) (course-to b)
                          (course-day b) a-day
                          (course-from b) a-from
                          (course-to b) a-to)
                    (format t "ok swap 2~%"))
                  (format t "err NOTFOUND~%")))))))

(defun handle-rename (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((old (nth 0 args))
            (new (nth 1 args)))
        (if (or (not (valid-word-p old))
                (not (valid-word-p new)))
            (format t "err BADARG~%")
            (let ((c (find-course old)))
              (if c
                  (if (find-course new)
                      (format t "err DUP~%")
                      (progn (setf (course-name c) new)
                             (format t "ok rename 1~%")))
                  (format t "err NOTFOUND~%")))))))

(defun handle-duplicate (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((name (nth 0 args))
            (newname (nth 1 args)))
        (if (or (not (valid-word-p name))
                (not (valid-word-p newname)))
            (format t "err BADARG~%")
            (let ((c (find-course name)))
              (if c
                  (cond ((find-course newname) (format t "err DUP~%"))
                        ((>= (length *courses*) 20) (format t "err LIMIT~%"))
                        (t
                         (push (make-course newname
                                            (course-teacher c)
                                            (course-loc c)
                                            (course-day c)
                                            (course-from c)
                                            (course-to c))
                               *courses*)
                         (format t "ok duplicate 1~%")))
                  (format t "err NOTFOUND~%")))))))

(defun handle-trim (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((day-s (nth 0 args))
            (to-s (nth 1 args)))
        (let ((day (parse-int day-s))
              (to (parse-int to-s)))
          (if (or (not day) (< day 1) (> day 7)
                  (not to) (< to 1) (> to 12))
              (format t "err BADARG~%")
              (let ((count 0))
                (dolist (c *courses*)
                  (when (and (= (course-day c) day)
                             (<= (course-from c) to)
                             (< to (course-to c)))
                    (setf (course-to c) to)
                    (incf count)))
                (format t "ok trim ~d~%" count)))))))

(defun handle-clear (args)
  (if (/= (length args) 1)
      (format t "err BADARG~%")
      (let ((day-s (first args)))
        (let ((day (parse-int day-s)))
          (if (or (not day) (< day 1) (> day 7))
              (format t "err BADARG~%")
              (let ((old-len (length *courses*)))
                (setf *courses* (remove-if (lambda (c) (= (course-day c) day)) *courses*))
                (format t "ok clear ~d~%" (- old-len (length *courses*)))))))))

(defun handle-shift (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((day-s (nth 0 args))
            (delta-s (nth 1 args)))
        (let ((day (parse-int day-s))
              (delta (parse-int delta-s)))
          (if (or (not day) (< day 1) (> day 7)
                  (not delta))
              (format t "err BADARG~%")
              (let ((new-day (+ day delta)))
                (if (or (< new-day 1) (> new-day 7))
                    (format t "err BADARG~%")
                    (let ((count 0))
                      (dolist (c *courses*)
                        (when (= (course-day c) day)
                          (setf (course-day c) new-day)
                          (incf count)))
                      (format t "ok shift ~d~%" count)))))))))

(defun handle-copy (args)
  (if (/= (length args) 2)

                            (t
                             (push (make-course newname
                                                (course-teacher c)
                                                (course-loc c)
                                                day
                                                (course-from c)
                                                (course-to c))
                                   *courses*)
                             (format t "ok copy 1~%"))))

(defun handle-merge (args)
  (if (/= (length args) 2)
      (format t "err BADARG~%")
      (let ((a-name (nth 0 args))
            (b-name (nth 1 args)))
        (if (or (not (valid-word-p a-name))

            (format t "err BADARG~%")
            (let ((a (find-course a-name))
                  (b (find-course b-name)))
              (if (and a b)
                  (if (string= a-name b-name)
                      (format t "err BADARG~%")
                      (progn
                        (setf (course-teacher a) (course-teacher b)
                              (course-loc a) (course-loc b))
                        (setf *courses* (remove b *courses* :test #'eq))
                        (format t "ok merge 2~%")))
                  (format t "err NOTFOUND~%")))))))

(defun handle-sort (args)
  (if (not (null args))
      (format t "err BADARG~%")
      (progn
        (setf *courses* (sort (copy-list *courses*) #'course-name<))
        (format t "ok sort 0~%"))))

(defun handle-filter (args)
  (if (/= (length args) 1)
      (format t "err BADARG~%")
      (let ((day-s (first args)))
        (let ((day (parse-int day-s)))
          (if (or (not day) (< day 1) (> day 7))
              (format t "err BADARG~%")
              (let ((count (count-if (lambda (c) (= (course-day c) day)) *courses*)))
                (format t "ok filter ~d~%" count)))))))

(defun handle-count (args)
  (if (not (null args))
      (format t "err BADARG~%")
      (format t "ok count ~d~%" (length *courses*))))

(defun process-command (cmd)
  (let* ((words (split-words cmd))
         (verb (and words (string-downcase (first words))))
         (args (rest words)))
    (cond ((null words) (format t "err BADARG~%"))
          ((string= verb "add") (handle-add args))
          ((string= verb "del") (handle-del args))
          ((string= verb "edit") (handle-edit args))
          ((string= verb "move") (handle-move args))
          ((string= verb "swap") (handle-swap args))
          ((string= verb "rename") (handle-rename args))
          ((string= verb "duplicate") (handle-duplicate args))
          ((string= verb "trim") (handle-trim args))
          ((string= verb "clear") (handle-clear args))
          ((string= verb "shift") (handle-shift args))
          ((string= verb "copy") (handle-copy args))
          ((string= verb "merge") (handle-merge args))
          ((string= verb "sort") (handle-sort args))
          ((string= verb "filter") (handle-filter args))
          ((string= verb "count") (handle-count args))
          (t (format t "err BADARG~%")))))

(defun command-string-p (s)
  (let ((words (split-words s)))
    (and words
         (member (string-downcase (first words))
                 '("add" "del" "edit" "move" "swap" "rename" "duplicate"
                   "trim" "clear" "shift" "copy" "merge" "sort" "filter" "count")
                 :test #'string=))))

(defun get-commands ()
  (let* ((argv (cdr sb-ext:*posix-argv*))
         (tail (member-if #'command-string-p argv)))
    tail))

(defun main ()
  (let ((commands (get-commands)))
    (dolist (cmd commands)
      (process-command cmd
